:language 1
:title "Alternative perspectives"
:output outcome
= focus "Generate plausible interpretations rather than reassurance."
!use therapy.ready as readiness {
    :with focus
}
?branch {
    :when readiness.decision.value == "begin" -> explore
    :when readiness.decision.value == "support" -> support
    :else -> declined
}
explore:
!ask step1 {
    What happened and how did you first interpret it?
}
!ask step2 {
    What are two other plausible explanations?
}
!ask step3 {
    What observations distinguish them, and what remains unknown?
}
!ask step4 {
    Which perspective seems fair and useful? What action follows?
}
!show {
    Your reflection, in your own words:

    1. {{step1}}
    2. {{step2}}
    3. {{step3}}
    4. {{step4}}

    You may leave this here. Nothing needs to feel resolved.
}
= outcome "reflected"
-> end
support:
!show {
    Stop the exercise. If anyone is in immediate danger, contact local emergency services. Reach a trusted person or qualified crisis support if you can. This exercise cannot provide emergency care.
}
= outcome "support_requested"
-> end
declined:
!show {
    We can leave this here. You do not need to explain or finish.
}
= outcome "declined"
-> end
